/**
 * @brief Opens the Linux kernel configuration menu using menuconfig.
 *
 * This command launches the Linux kernel configuration interface
 * through the Buildroot build system. It allows the user to customize
 * kernel options such as drivers, architecture, file systems, and more.
 *
 * @command
 * BR2_EXTERNAL=../buildroot-external-microchip make linux-menuconfig
 * BR2_EXTERNAL=../buildroot-external-microchip make linux-reconfigure
 * BR2_EXTERNAL=../buildroot-external-microchip make linux
 *
 * @note Make sure the Buildroot environment is properly configured
 * and the external directory contains the required overlays.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Opens the main Buildroot configuration menu using menuconfig.
 *
 * This command allows configuring all aspects of the embedded system,
 * including the toolchain, packages, kernel options, root filesystem,
 * and hardware-specific settings.
 *
 * @command
 * BR2_EXTERNAL=../buildroot-external-microchip make menuconfig
 *
 * @note Changes are saved into the .config file inside the output/ directory.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Removes all generated Device Tree Blob (DTB) files.
 *
 * This command cleans the generated .dtb files from the output images
 * directory. It is useful when forcing DTB regeneration after modifying
 * .dts or .dtsi files.
 *
 * @command
 * rm -f output/images/*.dtb
 *
 * @warning Make sure not to accidentally delete other important files.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Cleans the Linux kernel build directory.
 *
 * This command removes all files generated during kernel compilation,
 * allowing a clean rebuild from scratch. It is useful when inconsistencies
 * are suspected or major configuration changes were made.
 *
 * @command
 * BR2_EXTERNAL=../buildroot-external-microchip make linux-dirclean
 *
 * @note This does not affect the Buildroot configuration or other packages.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Builds the complete embedded Linux system using Buildroot.
 *
 * This command runs the full Buildroot compilation process,
 * including the toolchain, Linux kernel, root filesystem,
 * and all selected packages. It uses the configuration
 * defined in the .config file.
 *
 * @command
 * BR2_EXTERNAL=../buildroot-external-microchip make
 *
 * @note Ensure the system has been properly configured before building.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Loads the default configuration for the SAMA5D2 Xplained board.
 *
 * This command initializes the Buildroot `.config` file using
 * the predefined `atmel_sama5d2_xplained_mmc_defconfig`,
 * which contains default settings for the ARM architecture,
 * Linux kernel, root filesystem, and supported peripherals.
 *
 * @command
 * BR2_EXTERNAL=../buildroot-external-microchip make atmel_sama5d2_xplained_mmc_defconfig
 *
 * @note This step is required before running `make` to build the system.
 * @warning This command overwrites any existing `.config` configuration.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Extracts the generated Buildroot root filesystem archive.
 *
 * This command copies the generated rootfs.tar image from the Buildroot
 * output directory and extracts its contents into the current directory.
 * It is useful for inspecting or modifying the generated root filesystem.
 *
 * @commands
 * sudo cp /home/vboxuser/Documents/GESBC-5200/buildroot/buildroot-mchp/output/images/rootfs.tar .
 * sudo tar -xpf rootfs.tar
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Mounts a USB storage device into the system.
 *
 * These commands create a mount point and mount a FAT-formatted
 * USB device to the `/mnt/usb` directory.
 *
 * @commands
 * mkdir -p /mnt/usb
 * mount -t vfat /dev/sda1 /mnt/usb
 * mount -t vfat /dev/sdb1 /mnt/usb
 *
 * @note Replace `/dev/sda1` or `/dev/sdb1` with the correct device node.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Searches recursively for a symbol or keyword in source files.
 *
 * This command searches all `.c` and `.h` files recursively
 * for occurrences of the specified keyword.
 *
 * @command
 * grep -rnw --include="*.c" --include="*.h" . -e 'vmalloc'
 *
 * @note Useful for locating kernel functions, symbols, or APIs.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Configures Linux kernel boot arguments in U-Boot.
 *
 * This command sets the kernel boot arguments used during system startup.
 * It configures the serial console, root filesystem device,
 * read/write mode, and waits for the root device to become available.
 *
 * @command
 * setenv bootargs 'console=ttyS0,115200 root=/dev/mmcblk0p2 rw rootwait'
 *
 * @note These settings are typically used in embedded Linux systems booted from MMC/SD storage.
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Builds the Docker image for the ARM/Buildroot development environment.
 *
 * This command builds a Docker image containing all dependencies
 * required for cross-compiling the embedded Linux system using Buildroot.
 *
 * @command
 * sudo docker build  -t buildroot-arm-env -f docker/Dockerfile .
 */
/////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////////

/**
 * @brief Runs the Buildroot Docker development container.
 *
 * This command launches an interactive Docker container
 * and mounts the local project directory into the container
 * workspace at `/workspace/project`.
 *
 * @command
 * sudo docker run -it --rm -u $(id -u):$(id -g) -v $(pwd):/workspace/project buildroot-arm-env
 *
 * @note The `--rm` option automatically removes the container after exit.
 */
